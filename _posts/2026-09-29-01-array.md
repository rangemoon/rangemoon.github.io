## 一. 理论
数组在内存中的存储方式是：存放在连续内存空间上的相同类型数据的集合

1. 数组可以方便的通过下标索引的方式获取到下标对应的数据
2. 数组下标都是从0开始的，数组内存空间地址都是连续的
   - 因为数组的内存空间是连续的，删除和添加元素时，难免移动其他元素地址

3. C++中，需要注意vector和array(vector的底层实现是array；严格来说，vector是容器，不是数组)的区别
   - 数组的元素是不能删除的，只能覆盖
4. 二维数组在内存的空间地址是连续的

   ```cpp
   void test_arr() {
    int array[2][3] = {
   ```

## 二、二分查找
二分查找存在两种写法，取决于target是在一个什么样的区间内

1. target是在一个左闭右闭区间，也就是[Left, Right]
   - while (left <= right),因为left == right有意义
   - if (nums[middle] > right) right = middle - 1，因为nums[middle] 一定不是middle

   ```cpp
   int search(vector<int>& nums, int target) {
    int left = 0;
    int right = nums.size() - 1; // target in [left, right]
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (nums[mid] > target) right = mid - 1;
        else if (nums[mid] < target) left = mid + 1;
        else return mid;
    }
    return -1;
   }
   ```

2. target是在一个左闭右开区间，也就是[Left, Right)
   - while (left < right),因为left == right在区间[Left, Right)是没有意义的
   - if (nums[mid] > target) right = mid，因为当前nums[mid]不等于target，去左区间继续寻找。

   ```cpp
   int search(vector<int>& nums, int target) {
    int left = 0, right = nums.size();
    // target in [left, right)
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (nums[mid] > target) right = mid;
        else if (nums[mid] < target) left = mid + 1;
        return mid;
    }
    return -1;
   }
   ```

   ```cpp
    vector<int> ans(n);

        if (nums[i] * nums[i] > nums[j] * nums[j]) {
            ans[pos] = nums[i] * nums[i];
   ```

## 四、长度最小的子数组(滑动窗口)
1. 所谓滑动窗口就是不断调整子序列的起始位置和终止位置，从而得出我们想要的结果

### leetcode

1. leetcode 209. 长度最小的子数组

   ```cpp
   int minSubArrayLen(int target, vector<int>& nums) {
    int res = INT_MAX, subLen = 0;
    int l = 0;
    int sum = 0;
    for (int r = 0; r < nums.size(); ++r) {
        sum += nums[r];
        while (sum >= target) {
            subLen = (r - l + 1);
            res = min(res, subLen);
            sum -= nums[l++];
        }
    }
    return res == INT_MAX ? 0 : res;
   }
   ```

   - 时间复杂度：O(n)；空间复杂度：O(1)
   - 每个元素在滑动窗口后进来操作一次，出去再操作一次，每个元素都是被操作两次，所以事件复杂度是2 * n也就是O(n)

2. leetcode 904. 水果成蓝

   ```cpp
   int totalFruit(vector<int>& fruits) {
    int l = 0, res = 0;
    unordered_map<int, int> ft;
    for (int r = 0; r < fruits.size(); ++r) {
        ft[fruits[r]]++;
        while (ft.size() > 2) {
            auto it = ft.find(fruits[l]);
            // move left one by one
            it->second--;
            if (it->second == 0) {
                ft.erase(it);
            }
            l++;
        }
        res = max(res, r - l + 1);
    }
    return res;
   }
   ```

   ```cpp
   string minWindow(string s, string t) {
    int n = t.length(), m = s.length();
    if (n > m || n == 0) return "";
    int missing = n; // 还缺少多个个t中的字符
    int minLen = INT_MAX;
    int l = 0, start = 0;
    unordered_map<char, int> charMap;
    for(char ch : t) {
        charMap[ch]++;
    }
    for (int r = 0; r < m; ++r) {
        if (charMap[s[r]] > 0) {
            missing--;
        }
        charMap[s[r]]--;
        while (missing == 0) {
            if (r - l + 1 < minLen) {
                minLen = r - l + 1;
                start = l;
            }
            charMap[s[l]]++;
            if (charMap[s[l]] > 0)    // 说明这个字符是必需的
                missing++;
            l++;
        }
    }
    return minLen == INT_MAX ? "" : s.substr(start, minLen);
   }
   ```

## 五、螺旋矩阵2(模拟)
### leetcode
1. leetcode 59. 螺旋矩阵2
- 给你一个正整数 n ，生成一个包含 1 到 n^2 所有元素，且元素按顺时针顺序螺旋排列的 n x n 正方形矩阵 matrix

    ```cpp
    vector<vector<int>> generateMatrix(int n) {
        int num = 1;
        vector<vector<int>> matrix(n, vector<int>(n));
        int left = 0, right = n - 1, top = 0, bottom = n - 1;
        while (left <= right && top <= bottom) {
            for (int c = left; c <= right; ++c) {
                matrix[top][c] = num;
                num++;
            }
            for (int r = top + 1; r <= bottom; ++r) {
                matrix[r][right] = num;
                num++;
            }
            if (left < right && top < bottom) {
                for (int c = right - 1; c > left; --c) {
                    matrix[bottom][c] = num;
                    num++;
                }
                for (int r = bottom; r > top; --r) {
                    matrix[r][left] = num;
                    num++;
                }
            }
            left++;
            right--;
            top++;
            bottom--;
        }
        return matrix;
    }
    ```

2. leetcode 54. 螺旋矩阵
- 给你一个 m 行 n 列的矩阵 matrix ，请按照**顺时针螺旋顺序**，返回矩阵中的所有元素。

    ```cpp
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        if (matrix.size() == 0 || matrix[0].size() == 0) {
            return {};
        }
        int left = 0, right = matrix[0].size() - 1, top = 0, bottom = matrix.size() - 1;
        vector<int> res;
        while (left <= right && top <= bottom) {
            for (int c = left; c <= right; ++c) {
                res.push_back(matrix[top][c]);
            }
            for (int r = top + 1; r <= bottom; ++r) {
                res.push_back(matrix[r][right]);
            }
            if (left < right && top < bottom) {
                for (int c = right - 1; c > left; --c) {
                    res.push_back(matrix[bottom][c]);
                }
                for (int r = bottom; r > top; --r) {
                    res.push_back(matrix[r][left]);
                }
            }
            left++;
            right--;
            top++;
            bottom--;
        }
        return res;
    }
    ```

## 六、区间和(前缀和)
1. 前缀和的思想是重复利用计算过的子数组之和，从而降低区间查询需要累加计算的次数。
2. 前缀和在涉及计算**区间和**的问题时非常有用！
- 给定一个整数数组Array，计算该数组中指定区间内的元素总和
    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    int main() {
        int n; cin >> n;
        vector<int> nums(n);
        vector<int> p(n);
        int sum = 0;
        for (int i = 0; i < n; ++i) {
            cin >> nums[i];
            sum += nums[i];
            p[i] = sum;
        }
        int a, b;
        while (cin >> a && cin >> b) {
            if (a == 0) cout << p[b] << "\n";
            else cout << p[b] - p[a - 1] << "\n";
        }
        return 0;
    }
    ```
